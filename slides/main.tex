\setkicker{02 \quad·\quad VARIABLE SELECTION}

\begin{frame}{LASSO for a yes/no outcome}
  \vspace{1mm}\centering
  \keyformula{$\displaystyle \begin{gathered}
    \min_{b_0,\,\beta}\;\; \underbrace{-\frac{1}{n}\sum_{i=1}^{n}\Big[y_i\log p_i + (1-y_i)\log(1-p_i)\Big]}_{\text{fit to the data}} \;+\; \underbrace{\lambda\,\lVert\beta\rVert_1}_{\text{L1 penalty}}\\[1.5mm]
    \text{with}\;\; p_i = \frac{1}{1+e^{-(b_0 + x_i^\top\beta)}}\,:\; \text{firm } i\text{'s predicted bankruptcy probability}
  \end{gathered}$}\par\vspace{3mm}
  \raggedright
  \begin{points}
    \item The L1 penalty sets many coefficients to \textbf{exactly zero}.
    \item \textbf{One knob}, $\lambda$ (or $C = 1/n\lambda$), sets how many ratios survive.
  \end{points}
\end{frame}

\begin{frame}{Choosing the penalty}
  \vspace{2mm}\centering
  \includegraphics{lasso_path.pdf}\par\vspace{4mm}\raggedright
  \takeaway{Lowest cross-validated log-loss at $C \approx 0.18$: 43 of the 94 ratios are kept.}
\end{frame}
